-- ============================================================================
-- 성형외과 웹사이트 백엔드 / Oracle Database SQL
-- 기준 소스: clinic-backend/db.js, server.js, src/entity/*.js
-- DBMS: Oracle Database 12c 이상 (IDENTITY 지원)
-- 인코딩: UTF-8 BOM (Windows 메모장/SQL Developer 한글 호환)
-- 중요: db.js 는 synchronize: false 이므로 테이블 자동 생성되지 않음.
-- 중요: 신규/빈 스키마에서 최초 1회 실행. 이미 테이블이 있다면 생성문 재실행 금지.
-- 중요: 각 파트는 통합본에 모두 포함됨. 통합본 실행 후 각 파트 중복 실행 금지.
-- ============================================================================

-- ============================================================================
-- 참고 / 실행 안내
-- ============================================================================
-- 1) Oracle SQL Developer에서 XEPDB1 등의 실제 프로젝트 사용자(스키마)로 접속.
-- 2) 신규 빈 스키마에서 01~07 순서대로 실행하거나 통합본을 '스크립트 실행(F5)'.
-- 3) 이미 일부 테이블이 있는 DB에는 CREATE TABLE을 그대로 다시 실행하지 말 것.
--    ORA-00955 발생 가능. 기존 데이터 보존 후 구조 차이를 확인할 것.
-- 4) db.js의 synchronize:false를 바꾸지 않아도 수동 SQL로 생성 가능.
-- 5) 주석과 테이블 정의는 소스 EntitySchema와 서버 API를 기준으로 작성.
-- 6) db.txt의 FOOTER_SETTINGS.shedules 오타와 camelCase 컬럼을 보정했고,
--    기존 db.txt 끝의 실행 불가능한 학습 메모는 제외함.
-- 7) src/entity/SiteThemes.js는 파일은 있지만 db.js의 entities에 등록되지
--    않았고 현재 server.js에서도 참조되지 않아 운영 생성 대상에서 제외함.
-- 8) FOREIGN KEY: 엔티티에 명시 관계가 없으며 비회원 상담 USER_ID 값은
--    '비회원'이 될 수 있으므로 임의의 외래키는 만들지 않았음.
-- 9) 첨부 파일은 정의용 SQL이며, 실행 테스트는 DB 접속 권한 부재로 미수행.
-- 10) Member.js / Consult.js 등 일부 엔티티의 default에 미리 붙인 작은따옴표는
--     TypeORM Oracle 드라이버의 기본값 따옴표 처리와 충돌할 수 있음.
--     본 SQL의 DEFAULT 값은 Oracle에서 정상 실행되는 표기로 통일했음.
-- ============================================================================

-- PART 01. 회원 및 인증
-- 사용 기능: 회원가입, 로그인, 회원 조회/정지/삭제, 비밀번호 재설정
-- 관련 소스: src/entity/Member.js

-- MEMBER_TB 테이블에 CHART_NO 컬럼 추가(병원 환자차트번호 필수)
ALTER TABLE MEMBER_TB ADD CHART_NO VARCHAR2(20) UNIQUE;

CREATE TABLE MEMBER_TB (
    USER_IDX       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    CHART_NO       VARCHAR2(20) UNIQUE,
    USER_NAME      VARCHAR2(50) NOT NULL,
    USER_ID        VARCHAR2(50) NOT NULL,
    USER_PW        VARCHAR2(100) NOT NULL,
    EMAIL          VARCHAR2(100) NOT NULL,
    IS_MAIL_AGREED CHAR(1) DEFAULT 'N' NOT NULL,
    PHONE          VARCHAR2(20) NOT NULL,
    RESIDENT_NUM   VARCHAR2(255) NOT NULL,
    ZIPCODE        VARCHAR2(10),
    ADDRESS1       VARCHAR2(200),
    ADDRESS2       VARCHAR2(200),
    IS_SNS_AGREED  CHAR(1) DEFAULT 'N' NOT NULL,
    GENDER         VARCHAR2(10),
    IS_ADMIN       NUMBER DEFAULT 0 NOT NULL,
    REG_DATE       DATE DEFAULT SYSDATE NOT NULL,
    STATUS         VARCHAR2(20) DEFAULT '정상',
    CONSTRAINT UQ_MEMBER_TB_USER_ID UNIQUE (USER_ID)
);

COMMENT ON TABLE MEMBER_TB IS '회원 계정과 관리자 구분 정보';
COMMENT ON COLUMN MEMBER_TB.USER_IDX IS '자동 증가 회원 고유번호';
COMMENT ON COLUMN MEMBER_TB.USER_ID IS '로그인 아이디 (중복 불가)';
COMMENT ON COLUMN MEMBER_TB.USER_PW IS 'bcrypt 해시 비밀번호 저장';
COMMENT ON COLUMN MEMBER_TB.RESIDENT_NUM IS '회원 가입 시 수집되는 주민등록번호 관련 데이터: 별도 보호조치 필요';
COMMENT ON COLUMN MEMBER_TB.IS_ADMIN IS '0 일반회원 / 1 관리자';
COMMENT ON COLUMN MEMBER_TB.STATUS IS '회원 상태: 정상 또는 정지';

-- PART 04. 팝업 관리
-- 관련 소스: PopupSetting.js, Popup.js
-- 테이블: POPUP_SETTING, POPUP_TB
-- ============================================================================

-- 04-1. 팝업 노출 설정
CREATE TABLE POPUP_SETTING (
    ID         NUMBER PRIMARY KEY,
    MAX_POPUPS NUMBER DEFAULT 1 NOT NULL
);

COMMENT ON TABLE POPUP_SETTING IS '한 번에 표시할 최대 팝업 수 / ID = 1';

-- 04-2. 팝업 콘텐츠
-- START_DATE / END_DATE는 원본 엔티티와 동일하게 문자열 타입임.
CREATE TABLE POPUP_TB (
    POPUP_IDX       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    TITLE           VARCHAR2(200) NOT NULL,
    LINK            VARCHAR2(500),
    FILE_NAME       VARCHAR2(255) NOT NULL,
    START_DATE      VARCHAR2(50) NOT NULL,
    END_DATE        VARCHAR2(50) NOT NULL,
    USE_TODAY_CLOSE CHAR(1) DEFAULT 'Y' NOT NULL
);

COMMENT ON TABLE POPUP_TB IS '팝업 이미지, 연결 URL 및 표시 기간';
COMMENT ON COLUMN POPUP_TB.USE_TODAY_CLOSE IS '오늘 하루 보지 않기 사용 여부 (Y/N)';

-- 초기 설정: 기존 데이터가 없어야만 ID=1, MAX_POPUPS=1 삽입.
-- 이미 ID=1 설정이 존재하면 그대로 유지함.
MERGE INTO POPUP_SETTING p
USING (SELECT 1 AS ID, 1 AS MAX_POPUPS FROM DUAL) s
   ON (p.ID = s.ID)
WHEN NOT MATCHED THEN
    INSERT (ID, MAX_POPUPS) VALUES (s.ID, s.MAX_POPUPS);

COMMIT;

-- ============================================================================
-- PART 09. ADMIN_RESERVATION (관리자 예약 및 대기/진료 현황)
-- 관련 소스: AdminReservation.js (EntitySchema)
-- 테이블: ADMIN_RESERVATION_TB
-- ============================================================================

CREATE TABLE ADMIN_RESERVATION_TB (
    RESERVATION_IDX NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    CHART_NO VARCHAR2(20) NOT NULL,
    TREATMENT_IDX NUMBER DEFAULT NULL,
    RESERVATION_DATE DATE NOT NULL,
    RESERVATION_TIME VARCHAR2(10) NOT NULL,
    STATUS VARCHAR2(20) DEFAULT 'RESERVED' NOT NULL,
    MEMO VARCHAR2(255) DEFAULT NULL,
    CREATED_AT DATE DEFAULT SYSDATE NOT NULL,
    UPDATED_AT DATE DEFAULT SYSDATE NOT NULL,

    CONSTRAINT chk_admin_reservation_status CHECK (
        STATUS IN ('RESERVED', 'WAITING', 'CONSULTING', 'TREATING', 'COMPLETED', 'CANCELLED')
    ),

    CONSTRAINT fk_admin_reservation_chart_no
        FOREIGN KEY (CHART_NO)
        REFERENCES MEMBER_TB(CHART_NO)
        ON DELETE CASCADE,

    CONSTRAINT fk_admin_reservation_treatment
        FOREIGN KEY (TREATMENT_IDX)
        REFERENCES TREATMENT_TB(TREATMENT_IDX)
        ON DELETE SET NULL
);

-- 인덱스 추가
CREATE INDEX idx_admin_reservation_date_status
ON ADMIN_RESERVATION_TB(RESERVATION_DATE, STATUS);

COMMENT ON TABLE ADMIN_RESERVATION_TB IS '관리자 등록 병원 예약 및 당일 대기·진료 현황 정보';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.RESERVATION_IDX IS '예약/접수 고유번호 (자동 증가)';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.CHART_NO IS '환자 차트 번호 (MEMBER_TB FK)';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.TREATMENT_IDX IS '시술권 고유번호 (TREATMENT_TB FK)';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.RESERVATION_DATE IS '예약 및 진료 날짜';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.RESERVATION_TIME IS '예약 및 접수 시간 (HH:MM 형식)';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.STATUS IS '진료 상태 (RESERVED: 예약, WAITING: 대기, CONSULTING: 상담, TREATING: 치료중, COMPLETED: 완료, CANCELLED: 취소)';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.MEMO IS '환자 증상 및 메모 내용';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.CREATED_AT IS '예약/접수 생성일시';
COMMENT ON COLUMN ADMIN_RESERVATION_TB.UPDATED_AT IS '예약/접수 수정일시';

ALTER TABLE ADMIN_RESERVATION_TB

COMMENT ON COLUMN ADMIN_RESERVATION_TB.TREATMENT_IDX

ON DELETE SET NULL;

-- PART 10. TREATMENT (시술 및 패키지(회원권)정보)
-- 관련 소스: Treatment.js 
-- 테이블: TREATMENT_TB
-- ============================================================================

CREATE TABLE TREATMENT_TB (
    TREATMENT_IDX NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    CHART_NO VARCHAR2(20) NOT NULL,
    TREATMENT_NAME VARCHAR2(100) NOT NULL, -- 시술명 (예: 레이저 토닝 10회권)
    TOTAL_COUNT NUMBER(3) NOT NULL,       -- 총 횟수
    REMAINING_COUNT NUMBER(3) NOT NULL,   -- 남은 횟수
    STATUS VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL, -- 상태 (ACTIVE: 사용중, COMPLETED: 소진완료, EXPIRED: 기간만료)
    EXPIRE_DATE DATE DEFAULT NULL,        -- 유효기간 (선택사항)
    CREATED_AT DATE DEFAULT SYSDATE NOT NULL,
    UPDATED_AT DATE DEFAULT SYSDATE NOT NULL,
    
    -- 상태값 제약조건
    CONSTRAINT chk_treatment_status CHECK (
        STATUS IN ('ACTIVE', 'COMPLETED', 'EXPIRED')
    ),
    
    -- 회원(MEMBER_TB) 테이블의 차트번호와 외래키 연결
    CONSTRAINT fk_treatment_chart_no 
        FOREIGN KEY (CHART_NO) 
        REFERENCES MEMBER_TB(CHART_NO) 
        ON DELETE CASCADE
);

-- 테이블 코멘트
COMMENT ON TABLE TREATMENT_TB IS '환자 보유 시술 및 패키지(회원권) 관리 정보';

-- 컬럼 코멘트
COMMENT ON COLUMN TREATMENT_TB.TREATMENT_IDX IS '시술권 고유번호 (자동 증가)';
COMMENT ON COLUMN TREATMENT_TB.CHART_NO IS '환자 차트 번호 (MEMBER_TB FK)';
COMMENT ON COLUMN TREATMENT_TB.TREATMENT_NAME IS '시술 상품명 (예: 레이저 토닝 10회권)';
COMMENT ON COLUMN TREATMENT_TB.TOTAL_COUNT IS '총 시술 가능 횟수';
COMMENT ON COLUMN TREATMENT_TB.REMAINING_COUNT IS '남은 잔여 횟수';
COMMENT ON COLUMN TREATMENT_TB.STATUS IS '시술권 상태 (ACTIVE: 사용중, COMPLETED: 소진완료, EXPIRED: 기간만료)';
COMMENT ON COLUMN TREATMENT_TB.EXPIRE_DATE IS '시술권 유효기간 (선택사항)';
COMMENT ON COLUMN TREATMENT_TB.CREATED_AT IS '시술권 등록일시';
COMMENT ON COLUMN TREATMENT_TB.UPDATED_AT IS '시술권 수정일시';
